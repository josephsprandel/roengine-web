"""Clay's breath envelope for the page (META.env): RMS of the finished mono mix in 10 ms frames, smoothed
over 200 ms, normalised to 1. The page only reads it relative to Clay's own 98th percentile (the dust
brightens with the breath), so the shape is what matters. Defined here 2026-10-03; the earlier one wasn't saved."""
import numpy as np


def envelope(st, sr=44100):
    m = st.mean(1)
    n = len(m) // 441
    rms2 = (m[:n * 441].reshape(n, 441) ** 2).mean(1)
    r = np.sqrt(np.convolve(rms2, np.ones(20) / 20, mode='same'))
    return [round(float(v), 3) for v in r / r.max()]
